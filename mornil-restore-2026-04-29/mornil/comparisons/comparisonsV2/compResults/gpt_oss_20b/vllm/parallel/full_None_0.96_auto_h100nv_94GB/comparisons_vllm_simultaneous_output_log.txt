def process_questions(input_file_path):
    # Allowed difficulty levels
    allowed_difficulties = {"Easy", "Medium", "Hard"}

    # Statistics
    total_questions = 0
    valid_questions = 0
    invalid_questions = 0

    difficulty_counts = defaultdict(int)
    category_counts = defaultdict(int)
    tag_counts = defaultdict(int)

    # Validation failures
    validation_failures = []

    # Prepare output files
    valid_questions_file_path = 'valid_questions.json'
    validation_report_file_path = 'validation_report.txt'

    # Open the output files
    try:
        valid_questions_file = open(valid_questions_file_path, 'w', encoding='utf-8')
        validation_report_file = open(validation_report_file_path, 'w', encoding='utf-8')
    except IOError as e:
        print(f"Error opening output files: {e}")
        sys.exit(1)

    # Write the opening of the JSON array
    valid_questions_file.write('[\n')

    # Flag to manage commas between JSON objects
    first_valid_question = True

    # Process the input JSON file
    try:
        with open(input_file_path, 'r', encoding='utf-8') as f:
            # Use ijson to parse the top-level array
            questions = ijson.items(f, 'item')
            for question in questions:
                total_questions += 1
                failures = []

                # Validation checks
                # 1. questionText is not empty
                question_text = question.get('questionText')
                if not question_text or not isinstance(question_text, str) or question_text.strip() == '':
                    failures.append('questionText is empty')

                # 2. answers array has at least one answer
                answers = question.get('answers')
                if not answers or not isinstance(answers, list) or len(answers) == 0:
                    failures.append('answers array is empty')
                else:
                    # 3. Exactly one answer has isCorrect == true
                    correct_answers = [ans for ans in answers if ans.get('isCorrect') is True]
                    if len(correct_answers) != 1:
                        failures.append(f'number of correct answers is {len(correct_answers)} (expected 1)')

                # 4. difficulty is one of allowed values
                difficulty = question.get('difficulty')
                if difficulty not in allowed_difficulties:
                    failures.append(f'difficulty "{difficulty}" is invalid')

                # 5. category is not empty
                category = question.get('category')
                if not category or not isinstance(category, str) or category.strip() == '':
                    failures.append('category is empty')

                # 6. tags array is not empty
                tags = question.get('tags')
                if not tags or not isinstance(tags, list) or len(tags) == 0:
                    failures.append('tags array is empty')

                if failures:
                    # Invalid question
                    invalid_questions += 1
                    question_id = question.get('questionId', 'Unknown')
                    validation_failures.append((question_id, failures))
                else:
                    # Valid question
                    valid_questions += 1

                    # Update statistics
                    difficulty_counts[difficulty] += 1
                    category_counts[category] += 1
                    for tag in tags:
                        tag_counts[tag] += 1

                    # Write to valid_questions.json
                    if not first_valid_question:
                        valid_questions_file.write(',\n')
                    else:
                        first_valid_question = False
                    json.dump(question, valid_questions_file, ensure_ascii=False)
    except FileNotFoundError:
        print(f"Input file {input_file_path} not found.")
        sys.exit(1)
    except ijson.JSONError as e:
        print(f"Error parsing JSON: {e}")
        sys.exit(1)
    except Exception as e:
        print(f"An error occurred: {e}")
        sys.exit(1)

        # Close the JSON array
        valid_questions_file.write('\n]\n')
        valid_questions_file.close()
        validation_report_file.close()

    # Write validation report
    try:
        with open(validation_report_file_path, 'w', encoding='utf-8') as report_file:
            for qid, reasons in validation_failures:
                report_file.write(f"{qid}: {', '.join(reasons)}\n")
    except IOError as e:
        print(f"Error writing validation report: {e}")
        sys.exit(1)

    # Output statistics
    print(f"Total questions processed: {total_questions}")

    for cat, count in category_counts.items():
        print(f"  {cat}: {count}")
    print("\nValid questions per tag:")
    for tag, count in tag_counts.items():
        print(f"  {tag}: {count}")
